function toISODate(dateStr) {
  if (!dateStr) return "";
  const d = new Date(dateStr);
  if (isNaN(d.getTime())) return "";
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
}

function fromISODate(isoStr) {
  if (!isoStr) return "";
  const d = new Date(isoStr + "T00:00:00");
  if (isNaN(d.getTime())) return "";
  return d.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" }).replace(/,/g, "");
}

function formatDateRange(start, end) {
  const startDate = new Date(start);
  const endDate = new Date(end);
  const fullFmt = { weekday: "short", day: "numeric", month: "short", year: "numeric" };
  const startDisplay = startDate.toLocaleDateString("en-GB", fullFmt).replace(",", "");
  const endDisplay = endDate.toLocaleDateString("en-GB", fullFmt).replace(",", "");
  return `${startDisplay} | ${endDisplay}`;
}

function formatToday() {
  const fullFmt = { weekday: "short", day: "numeric", month: "short", year: "numeric" };
  return new Date().toLocaleDateString("en-GB", fullFmt).replace(",", "");
}

function getWaitingHours(requestedAt) {
  const diffMs = new Date() - new Date(requestedAt);
  return diffMs / (1000 * 60 * 60);
}

function getWaitingColor(hours) {
  if (hours < 6) return COLORS.green;
  if (hours < 11) return COLORS.amber;
  return COLORS.red;
}

function formatWaitingTime(hours) {
  const wholeHours = Math.floor(hours);
  const minutes = Math.round((hours - wholeHours) * 60);
  return minutes > 0 ? `${wholeHours}h${String(minutes).padStart(2, "0")}min` : `${wholeHours}h`;
}

function getCotsMattresses(b) {
  if (typeof b.cots === "number") return b.cots;
  const countableGuests = b.adults + b.children;
  return Math.max(0, countableGuests - b.rooms * 2);
}

function getMinimumRooms(b) {
  const countableGuests = b.adults + b.children;
  return Math.max(1, Math.ceil(countableGuests / 3));
}

function isContactUnlocked(tripDate) {
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const trip = new Date(tripDate);
  trip.setHours(0, 0, 0, 0);
  const daysUntilTrip = (trip - today) / (1000 * 60 * 60 * 24);
  return daysUntilTrip <= 4;
}

function isUnderRoomed(b) {
  return b.rooms < getMinimumRooms(b);
}

function getAvailabilityStatus(boat, dateStr, allBookings, allBlockedDates, dateOpenState, dateTripPricing) {
  const isDateOpen = dateOpenState[`${boat}|${dateStr}`] === true;
  if (!isDateOpen) return { status: "closed" };

  const isTripTypeOpenAndPriced = (type) => {
    const entry = dateTripPricing[`${boat}|${dateStr}|${type}`];
    if (!entry?.tiers) return false;
    return Object.values(entry.tiers).some(tier => tier.open);
  };
  const openTypes = {
    "Day Cruise": isTripTypeOpenAndPriced("Day Cruise"),
    "Overnight Stay": isTripTypeOpenAndPriced("Overnight Stay"),
    "Night Stay": isTripTypeOpenAndPriced("Night Stay"),
  };

  const sameDayBookings = allBookings.filter(
    b => b.boat === boat && b.date === dateStr && b.status !== "cancelled" && b.status !== "deleted"
  );
  const sameDayDirectBookings = allBlockedDates.filter(
    b => b.boat === boat && b.date === dateStr && b.reason === "direct"
  );
  const hasDirectBooking = sameDayDirectBookings.length > 0 || sameDayBookings.some(b => b.isDirect);

  const isBooked = (type) => {
    const bookingTypeMap = { "Day Cruise": "Day cruise", "Overnight Stay": "Overnight stay", "Night Stay": "Night stay" };
    const realBooked = sameDayBookings.some(b => b.type === bookingTypeMap[type]);
    const directBooked = sameDayDirectBookings.some(b => b.tripType === type);
    return realBooked || directBooked;
  };

  const dayBooked = isBooked("Day Cruise");
  const overnightBooked = isBooked("Overnight Stay");
  const nightBooked = isBooked("Night Stay");

  const isVisible = {
    "Day Cruise": openTypes["Day Cruise"] || dayBooked || overnightBooked,
    "Overnight Stay": openTypes["Overnight Stay"] || overnightBooked || dayBooked || nightBooked,
    "Night Stay": openTypes["Night Stay"] || nightBooked || overnightBooked,
  };

  const circles = {
    "Day Cruise": isVisible["Day Cruise"] ? ((dayBooked || overnightBooked) ? "red" : "green") : null,
    "Overnight Stay": isVisible["Overnight Stay"] ? ((overnightBooked || dayBooked || nightBooked) ? "red" : "green") : null,
    "Night Stay": isVisible["Night Stay"] ? ((nightBooked || overnightBooked) ? "red" : "green") : null,
  };

  const openCircleColors = Object.values(circles).filter(c => c !== null);

  if (openCircleColors.length === 0) {
    return { status: "empty", circles, hasDirectBooking };
  }

  const allRed = openCircleColors.every(c => c === "red");
  const someRed = openCircleColors.some(c => c === "red");
  const background = allRed ? "red" : someRed ? "amber" : "green";

  return { status: "open", circles, background, hasDirectBooking };
}

function buildDefaultPricing(boatName) {
  const bhTiers = BOAT_BH_CONFIGS[boatName] || [BOAT_TOTAL_BH[boatName]];
  const pricing = {};
  TRIP_TYPES.forEach((type, typeIndex) => {
    const tiers = {};
    bhTiers.forEach(bh => {
      const basePerBH = 6000 + typeIndex * 2000;
      tiers[bh] = {
        base: basePerBH * bh,
        extraAdult: 700 + typeIndex * 100,
        extraChild: 350 + typeIndex * 50,
      };
    });
    pricing[type] = { open: true, tiers };
  });
  return pricing;
}

const BOOKING_TYPE_TO_AVAILABILITY_TYPE = {
  "Day cruise": "Day Cruise",
  "Overnight stay": "Overnight Stay",
  "Night stay": "Night Stay",
};

function buildBackfilledPricing(allBookings) {
  const seeded = {};
  allBookings.forEach(b => {
    const availabilityType = BOOKING_TYPE_TO_AVAILABILITY_TYPE[b.type];
    if (!availabilityType) return;
    const defaults = buildDefaultPricing(b.boat);
    const bhTiers = BOAT_BH_CONFIGS[b.boat] || [BOAT_TOTAL_BH[b.boat]];
    const bookedBH = defaults[availabilityType]?.tiers?.[b.rooms] ? b.rooms : bhTiers[0];
    const key = `${b.boat}|${b.date}|${availabilityType}`;
    const tiers = {};
    bhTiers.forEach(bh => {
      const base = defaults[availabilityType]?.tiers?.[bh];
      tiers[bh] = { ...base, open: bh === bookedBH };
    });
    seeded[key] = { tiers };
  });
  return seeded;
}

        const addDietRow = () => setAddBookingForm(prev => ({ ...prev, dietBreakdown: [...prev.dietBreakdown, { type: "Veg", count: 0 }] }));
        const removeDietRow = (index) => setAddBookingForm(prev => ({ ...prev, dietBreakdown: prev.dietBreakdown.filter((_, i) => i !== index) }));
        const overnightCheckOutTooEarly = form.type === "Overnight Stay" && form.checkIn && new Date(form.checkOut) < new Date(minCheckOutForOvernight(form.checkIn));
        const canConfirm = form.guest.trim() && form.phone.trim() && form.price !== "" && !dietMismatch && !roomsTooFew && !typeUnavailable[form.type] && !overnightCheckOutTooEarly;

        const handleConfirm = () => {
          if (!canConfirm) return;
          const isEditing = !!form.editingBookingId;
          const dbDates = [];
          if (form.type === "Overnight Stay") {
            let cursor = new Date(form.checkIn);
            const lastBlockedDate = new Date(form.checkOut);
            lastBlockedDate.setDate(lastBlockedDate.getDate() - 1);
            while (cursor <= lastBlockedDate) { dbDates.push(cursor.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" }).replace(/,/g, "")); cursor = new Date(cursor.getTime() + 24 * 60 * 60 * 1000); }
          } else { dbDates.push(form.checkIn); }
          const mappedType = form.type === "Day Cruise" ? "Day cruise" : form.type === "Overnight Stay" ? "Overnight stay" : "Night stay";
          if (isEditing) {
            setBookings(prev => prev.map(b => b.id !== form.editingBookingId ? b : { ...b, guest: form.guest.trim(), phone: form.phone.trim(), boat: form.boat, type: mappedType, date: form.checkIn, dateEnd: form.checkOut, adults: form.adults, children: form.children, kids: form.kids, rooms: form.rooms, cots: form.cots, dietBreakdown: form.dietBreakdown.filter(d => d.count > 0), specialRequests: form.specialRequests.trim() ? form.specialRequests.split(",").map(s => s.trim()).filter(Boolean) : [], price: Number(form.price), bookingSource: form.source }));
            dbDates.forEach(date => setDateOpenState(prev => ({ ...prev, [`${form.boat}|${date}`]: true })));
            setBlockedDates(prev => [...prev.filter(bd => !(bd.boat === form.originalBoat && form.originalDbDates.includes(bd.date) && bd.reason === "direct")), ...dbDates.map(date => ({ boat: form.boat, date, reason: "direct", tripType: form.type }))]);
          } else {
            const checkInDate = new Date(form.checkIn);
            const bookingId = `AB-${String(checkInDate.getDate()).padStart(2, "0")}${String(checkInDate.getMonth() + 1).padStart(2, "0")}${checkInDate.getFullYear()}-${String(Math.floor(1000 + Math.random() * 9000))}`;
            setBookings(prev => [...prev, { id: Date.now(), bookingId, guest: form.guest.trim(), phone: form.phone.trim(), boat: form.boat, type: mappedType, date: form.checkIn, dateEnd: form.checkOut, adults: form.adults, children: form.children, kids: form.kids, rooms: form.rooms, cots: form.cots, dietBreakdown: form.dietBreakdown.filter(d => d.count > 0), accessibility: "None", specialRequests: form.specialRequests.trim() ? form.specialRequests.split(",").map(s => s.trim()).filter(Boolean) : [], price: Number(form.price), isDirect: true, bookingSource: form.source }]);
            dbDates.forEach(date => setDateOpenState(prev => ({ ...prev, [`${form.boat}|${date}`]: true })));
            setBlockedDates(prev => [...prev, ...dbDates.map(date => ({ boat: form.boat, date, reason: "direct", tripType: form.type }))]);
          }
          setAddBookingFormOpen(false); setAddBookingForm(null); setAvailabilitySelection([]); setExpandedTripType(null); setPriceDrafts({}); setConfirmRatesError(null); setModifyBookingListOpen(false); setModifySelectedBookingId(null);
        };

        const fieldLabel = (text) => <div style={{ fontSize: 12, fontWeight: 600, color: COLORS.muted, marginBottom: 6 }}>{text}</div>;
        const inputStyle = { width: "100%", fontSize: 14, color: COLORS.navy, border: `1px solid ${COLORS.border}`, borderRadius: 10, padding: "10px 12px", boxSizing: "border-box" };

        return (
          <div style={{ position: "fixed", inset: 0, zIndex: 50, background: "rgba(15,23,42,0.5)", display: "flex", alignItems: "flex-end", justifyContent: "center" }} onClick={() => { setAddBookingFormOpen(false); setAddBookingForm(null); }}>
            <div onClick={(e) => e.stopPropagation()} style={{ width: "100%", maxWidth: 393, maxHeight: "88vh", overflowY: "auto", background: COLORS.white, borderRadius: "24px 24px 0 0", padding: "20px 18px 28px" }}>
              <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", marginBottom: 16 }}>
                <div style={{ fontSize: 17, fontWeight: 800, color: COLORS.navy }}>{form.editingBookingId ? "Edit Booking" : "Add Booking"}</div>
                <div onClick={() => { setAddBookingFormOpen(false); setAddBookingForm(null); }} style={{ cursor: "pointer", padding: 4 }}><Icon name="x" size={18} color={COLORS.muted} /></div>
              </div>
              <div style={{ display: "flex", flexDirection: "column", gap: 14 }}>
                <div>{fieldLabel("Booking Source")}<select style={inputStyle} value={form.source} onChange={(e) => updateField("source", e.target.value)}>{["Direct Booking", "Booking.com", "Airbnb", "Make My Trip", "Agoda", "Agent"].map(opt => <option key={opt} value={opt}>{opt}</option>)}</select></div>
                <div>{fieldLabel("Guest Name and Surname")}<input style={inputStyle} type="text" placeholder="e.g. Arjun Mehta" value={form.guest} onChange={(e) => updateField("guest", e.target.value)} /></div>
                <div>{fieldLabel("Houseboat")}<div style={{ ...inputStyle, background: COLORS.bg, color: COLORS.navy, fontWeight: 700, display: "flex", alignItems: "center", justifyContent: "space-between" }}><span>{form.boat}</span><select value={form.rooms} onChange={(e) => updateField("rooms", Number(e.target.value))} style={{ border: "none", background: "transparent", fontWeight: 700, color: COLORS.teal, fontSize: 14 }}>{bhTiers.map(bh => <option key={bh} value={bh}>{bh}BH</option>)}</select></div></div>
                <div>{fieldLabel("Cruise Type")}<select style={inputStyle} value={form.type} onChange={(e) => updateField("type", e.target.value)}>{TRIP_TYPES.map(t => <option key={t} value={t} disabled={typeUnavailable[t]}>{t}{typeUnavailable[t] ? " (already booked)" : ""}</option>)}</select>{typeUnavailable[form.type] && <div style={{ fontSize: 11, fontWeight: 600, color: COLORS.red, marginTop: 4 }}>{form.type} is already booked for this date.</div>}</div>
                <div style={{ display: "flex", gap: 10 }}>
                  <div style={{ flex: 1 }}>{fieldLabel("Check-in Date")}<input style={inputStyle} type="date" value={toISODate(form.checkIn)} onChange={(e) => updateField("checkIn", fromISODate(e.target.value))} /></div>
                  <div style={{ flex: 1 }}>{fieldLabel(form.type === "Day Cruise" ? "Check-out (same day)" : form.type === "Night Stay" ? "Check-out (next day)" : "Check-out (min. next day)")}<input style={{ ...inputStyle, background: isCheckOutLocked ? COLORS.bg : COLORS.white, color: isCheckOutLocked ? COLORS.muted : COLORS.navy }} type="date" value={toISODate(form.checkOut)} min={form.type === "Overnight Stay" ? toISODate(minCheckOutForOvernight(form.checkIn)) : undefined} disabled={isCheckOutLocked} onChange={(e) => updateField("checkOut", fromISODate(e.target.value))} /></div>
                </div>
                <div style={{ display: "flex", gap: 10 }}>
                  <div style={{ flex: 1 }}>{fieldLabel("Adults (13+)")}<input style={inputStyle} type="number" min="0" value={form.adults} onChange={(e) => updateField("adults", Math.max(0, parseInt(e.target.value, 10) || 0))} /></div>
                  <div style={{ flex: 1 }}>{fieldLabel("Children (6–12)")}<input style={inputStyle} type="number" min="0" value={form.children} onChange={(e) => updateField("children", Math.max(0, parseInt(e.target.value, 10) || 0))} /></div>
                  <div style={{ flex: 1 }}>{fieldLabel("Kids (0–5)")}<input style={inputStyle} type="number" min="0" value={form.kids} onChange={(e) => updateField("kids", Math.max(0, parseInt(e.target.value, 10) || 0))} /></div>
                </div>
                <div>{fieldLabel("Number of Rooms")}<input style={inputStyle} type="number" min="1" value={form.rooms} onChange={(e) => updateField("rooms", Math.max(1, parseInt(e.target.value, 10) || 1))} />{roomsTooFew && <div style={{ fontSize: 11, color: COLORS.red, marginTop: 4 }}>Needs at least {minRooms} room{minRooms !== 1 ? "s" : ""} for {countableGuests} guest{countableGuests !== 1 ? "s" : ""}</div>}</div>
                <div>{fieldLabel("Cot/Mattress")}<input style={inputStyle} type="number" min="0" value={form.cots} onChange={(e) => updateField("cots", Math.max(0, Number(e.target.value) || 0))} /><div style={{ fontSize: 11, color: COLORS.muted, marginTop: 4 }}>Suggested: {computedCots} (2 guests + 1 extra per room, kids excluded)</div></div>
                <div>
                  {fieldLabel("Diet per Guest")}
                  <div style={{ display: "flex", flexDirection: "column", gap: 8 }}>
                    {form.dietBreakdown.map((d, idx) => (
                      <div key={idx} style={{ display: "flex", gap: 8, alignItems: "center" }}>
                        <select style={{ ...inputStyle, flex: 1.4 }} value={d.type} onChange={(e) => updateDiet(idx, "type", e.target.value)}>{["Veg", "Non-veg", "Vegan", "Jain", "Halal"].map(opt => <option key={opt} value={opt}>{opt}</option>)}</select>
                        <input style={{ ...inputStyle, flex: 1 }} type="number" min="0" value={d.count} onChange={(e) => updateDiet(idx, "count", Math.max(0, parseInt(e.target.value, 10) || 0))} />
                        <div onClick={() => removeDietRow(idx)} style={{ cursor: "pointer", padding: 4, flexShrink: 0 }}><Icon name="x" size={14} color={COLORS.muted} /></div>
                      </div>
                    ))}
                    <div onClick={addDietRow} style={{ fontSize: 12, fontWeight: 700, color: COLORS.teal, cursor: "pointer" }}>+ Add diet type</div>
                    {dietMismatch && <div style={{ fontSize: 11, color: COLORS.red }}>Diet total ({dietTotal}) must equal adults + children ({countableGuests})</div>}
                  </div>
                </div>
                <div>{fieldLabel("Contact Number")}<input style={inputStyle} type="tel" placeholder="e.g. +91 98765 43210" value={form.phone} onChange={(e) => updateField("phone", e.target.value)} /></div>
                <div>{fieldLabel("Special Requests")}<input style={inputStyle} type="text" placeholder="e.g. Birthday cake, parking needed" value={form.specialRequests} onChange={(e) => updateField("specialRequests", e.target.value)} /></div>
                <div>{fieldLabel("Price (₹)")}<input style={inputStyle} type="number" min="0" placeholder="e.g. 18500" value={form.price} onChange={(e) => updateField("price", e.target.value)} /></div>
                <div onClick={handleConfirm} style={{ marginTop: 6, background: canConfirm ? COLORS.teal : COLORS.border, color: canConfirm ? COLORS.white : COLORS.muted, textAlign: "center", borderRadius: 12, padding: "14px 0", fontSize: 14, fontWeight: 700, cursor: canConfirm ? "pointer" : "default" }}>{form.editingBookingId ? "Save Changes" : "Confirm Booking"}</div>
              </div>
            </div>
          </div>
        );
      })()}

      {modifyBookingListOpen && (() => {
        const modifiableBookings = bookings.filter(b => b.boat === availabilityBoat && b.isDirect && b.status !== "cancelled" && b.status !== "deleted" && availabilitySelection.length > 0 && (() => { const sel = [...availabilitySelection].sort((a, b2) => a - b2); const min = sel[0], max = sel[sel.length - 1]; const dateStrFor = (day) => new Date(availabilityMonth.year, availabilityMonth.month, day).toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" }).replace(/,/g, ""); const selectedDates = []; for (let d = min; d <= max; d++) selectedDates.push(dateStrFor(d)); return selectedDates.includes(b.date); })());
        const selectedBooking = modifiableBookings.find(b => b.id === modifySelectedBookingId) || null;
        const closeModal = () => { setModifyBookingListOpen(false); setModifySelectedBookingId(null); setModifyDeleteConfirmOpen(false); };
        const handleStartEdit = () => {
          if (!selectedBooking) return;
          const availabilityType = selectedBooking.type === "Day cruise" ? "Day Cruise" : selectedBooking.type === "Overnight stay" ? "Overnight Stay" : "Night Stay";
          const originalDbDates = [];
          if (availabilityType === "Overnight Stay") {
            let cursor = new Date(selectedBooking.date);
            const lastBlockedDate = new Date(selectedBooking.dateEnd);
            lastBlockedDate.setDate(lastBlockedDate.getDate() - 1);
            while (cursor <= lastBlockedDate) { originalDbDates.push(cursor.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" }).replace(/,/g, "")); cursor = new Date(cursor.getTime() + 24 * 60 * 60 * 1000); }
          } else { originalDbDates.push(selectedBooking.date); }
          setAddBookingForm({ editingBookingId: selectedBooking.id, originalBoat: selectedBooking.boat, originalDbDates, source: selectedBooking.bookingSource || "Direct Booking", guest: selectedBooking.guest, boat: selectedBooking.boat, rooms: selectedBooking.rooms, type: availabilityType, checkIn: selectedBooking.date, checkOut: selectedBooking.dateEnd, adults: selectedBooking.adults, children: selectedBooking.children, kids: selectedBooking.kids, cots: typeof selectedBooking.cots === "number" ? selectedBooking.cots : Math.max(0, (selectedBooking.adults + selectedBooking.children) - selectedBooking.rooms * 2), dietBreakdown: selectedBooking.dietBreakdown && selectedBooking.dietBreakdown.length ? selectedBooking.dietBreakdown : [{ type: "Veg", count: 2 }], phone: selectedBooking.phone, specialRequests: (selectedBooking.specialRequests || []).join(", "), price: String(selectedBooking.price) });
          setAddBookingFormOpen(true); closeModal();
        };
        const handleConfirmDelete = () => {
          if (!selectedBooking) return;
          setBookings(prev => prev.map(b => b.id !== selectedBooking.id ? b : { ...b, status: "deleted" }));
          const availabilityType = selectedBooking.type === "Day cruise" ? "Day Cruise" : selectedBooking.type === "Overnight stay" ? "Overnight Stay" : "Night Stay";
          const datesToFree = [];
          if (availabilityType === "Overnight Stay") { let cursor = new Date(selectedBooking.date); const lastBlockedDate = new Date(selectedBooking.dateEnd); lastBlockedDate.setDate(lastBlockedDate.getDate() - 1); while (cursor <= lastBlockedDate) { datesToFree.push(cursor.toLocaleDateString("en-GB", { day: "numeric", month: "short", year: "numeric" }).replace(/,/g, "")); cursor = new Date(cursor.getTime() + 24 * 60 * 60 * 1000); } } else { datesToFree.push(selectedBooking.date); }
          setBlockedDates(prev => prev.filter(bd => !(bd.boat === selectedBooking.boat && datesToFree.includes(bd.date) && bd.reason === "direct" && bd.tripType === availabilityType)));
          closeModal();
        };

        return (
          <div style={{ position: "fixed", inset: 0, zIndex: 50, background: "rgba(15,23,42,0.5)", display: "flex", alignItems: "flex-end", justifyContent: "center" }}>
            <div style={{ width: "100%", maxWidth: 393, maxHeight: "88vh", overflowY: "auto", background: COLORS.white, borderRadius: "24px 24px 0 0", padding: "20px 18px 28px" }}>
              <div style={{ display: "flex", alignItems: "center", justifyContent: "space-between", marginBottom: 16 }}>
                <div style={{ fontSize: 17, fontWeight: 800, color: COLORS.navy }}>{selectedBooking ? "Booking Details" : "Modify Booking"}</div>
                <div onClick={closeModal} style={{ cursor: "pointer", padding: 4 }}><Icon name="x" size={18} color={COLORS.muted} /></div>
              </div>
              {!selectedBooking && (
                <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
                  {modifiableBookings.length === 0 && <div style={{ textAlign: "center", padding: "30px 0", color: COLORS.muted, fontSize: 13 }}>No added bookings found for this selection.</div>}
                  {modifiableBookings.map(b => (
                    <div key={b.id} onClick={() => setModifySelectedBookingId(b.id)} style={{ display: "flex", alignItems: "center", justifyContent: "space-between", border: `1px solid ${COLORS.border}`, borderRadius: 14, padding: "14px 16px", cursor: "pointer" }}>
                      <div><div style={{ fontWeight: 700, fontSize: 14, color: COLORS.navy }}>{b.guest}</div><div style={{ fontSize: 12, color: COLORS.muted, marginTop: 2 }}>{b.type === "Day cruise" ? "Day Cruise" : b.type === "Overnight stay" ? "Overnight Stay" : "Night Stay"} · {formatDateRange(b.date, b.dateEnd)}</div></div>
                      <Icon name="chevronRight" size={16} color={COLORS.muted} />
                    </div>
                  ))}
                </div>
              )}
              {selectedBooking && !modifyDeleteConfirmOpen && (
                <div>
                  <div onClick={() => setModifySelectedBookingId(null)} style={{ display: "flex", alignItems: "center", gap: 6, marginBottom: 14, cursor: "pointer", color: COLORS.teal, fontSize: 12, fontWeight: 600 }}><Icon name="arrowLeft" size={14} color={COLORS.teal} />Back to list</div>
                  <div style={{ border: `1px solid ${COLORS.border}`, borderRadius: 16, padding: 16, marginBottom: 18 }}>
                    <div style={{ display: "flex", alignItems: "center", gap: 8, flexWrap: "wrap", marginBottom: 10 }}><div style={{ fontWeight: 700, fontSize: 15, color: COLORS.navy }}>{selectedBooking.guest}</div><span style={{ fontSize: 10, fontWeight: 700, color: COLORS.navy, background: COLORS.tealLight, borderRadius: 999, padding: "2px 8px", textTransform: "uppercase", letterSpacing: "0.03em" }}>{selectedBooking.bookingSource || "Direct Booking"}</span></div>
                    {[["Cruise Type", selectedBooking.type === "Day cruise" ? "Day Cruise" : selectedBooking.type === "Overnight stay" ? "Overnight Stay" : "Night Stay"], ["Dates", formatDateRange(selectedBooking.date, selectedBooking.dateEnd)], ["Houseboat", selectedBooking.boat], ["Rooms", `${selectedBooking.rooms}BH`], ["Guests", `${selectedBooking.adults} adults · ${selectedBooking.children} children · ${selectedBooking.kids} kids`], ["Phone", selectedBooking.phone], ["Price", `₹${selectedBooking.price?.toLocaleString("en-IN")}`]].map(([label, value]) => <div key={label} style={{ display: "flex", justifyContent: "space-between", padding: "6px 0", borderTop: `1px solid ${COLORS.border}` }}><span style={{ fontSize: 12, color: COLORS.muted }}>{label}</span><span style={{ fontSize: 12, fontWeight: 600, color: COLORS.navy, textAlign: "right" }}>{value}</span></div>)}
                  </div>
                  <div style={{ display: "flex", gap: 10 }}>
                    <div onClick={handleStartEdit} style={{ flex: 1, textAlign: "center", borderRadius: 12, padding: "12px 0", border: `1px solid ${COLORS.teal}`, color: COLORS.teal, fontSize: 13, fontWeight: 700, cursor: "pointer" }}>Edit Booking</div>
                    <div onClick={() => setModifyDeleteConfirmOpen(true)} style={{ flex: 1, textAlign: "center", borderRadius: 12, padding: "12px 0", background: COLORS.red, color: COLORS.white, fontSize: 13, fontWeight: 700, cursor: "pointer" }}>Delete Booking</div>
                  </div>
                </div>
              )}
              {selectedBooking && modifyDeleteConfirmOpen && (
                <div style={{ textAlign: "center", padding: "10px 0" }}>
                  <div style={{ fontSize: 14, fontWeight: 600, color: COLORS.navy, marginBottom: 20 }}>Do you confirm the cancellation of this booking?</div>
                  <div style={{ display: "flex", gap: 10 }}>
                    <div onClick={() => setModifyDeleteConfirmOpen(false)} style={{ flex: 1, textAlign: "center", borderRadius: 12, padding: "12px 0", border: `1px solid ${COLORS.border}`, color: COLORS.navy, fontSize: 13, fontWeight: 700, cursor: "pointer" }}>No</div>
                    <div onClick={handleConfirmDelete} style={{ flex: 1, textAlign: "center", borderRadius: 12, padding: "12px 0", background: COLORS.red, color: COLORS.white, fontSize: 13, fontWeight: 700, cursor: "pointer" }}>Yes</div>
                  </div>
                </div>
              )}
            </div>
          </div>
        );
      })()}
    </div>
  );
}
